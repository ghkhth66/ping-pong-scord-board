                    * 편성 조 수: {saved_cfg.get('g')}조 _ <strong> (조당 {per_group_text}) </strong>
                    * 경기 방식: 단식 {saved_cfg.get('s_games')} / 복식 {saved_cfg.get('d_games')} (선승: {saved_cfg.get('set_count')}세트)
                    * 조 편성 방식: {saved_cfg.get('draw_method')}
                    """)

            if is_admin:
                st.warning("⚠️ 설정을 변경하면 진행 중인 조 편성과 대진표가 모두 초기화됩니다.")
                if st.button("🔓 설정 다시 하기", type="secondary"):
                    reset_draw_state(reset_config=True)
                    st.rerun()

        # ---------------------------------------------------------
        # 2️⃣ 설정이 확정되지 않은 경우 (입력 화면)
        # ---------------------------------------------------------
        else:
            if is_admin:
                st.markdown("<div style='margin-top: -20px;'></div>", unsafe_allow_html=True)
                c1, c2, c3, c4, c5 = st.columns([1.2, 1.2, 1.2, 1.2, 1.5], gap="small")

                with c1:
                    responsive_text(f"👥 조 구성", pc_size="20px", mobile_size="16px")
                    v_space(5)
                    default_g = max(1, min(20, saved_cfg.get("g", 4)))
                    max_groups = max(1, attendees_count)
                    safe_default_g = min(default_g, max_groups)

                    g_val = st.number_input("편성 조 수", min_value=1, max_value=max_groups, value=safe_default_g, disabled=not is_admin)
                    if g_val > 0:
                        avg = attendees_count // g_val
                        rem = attendees_count % g_val
                        if rem > 0:
                            st.info(f"👉 조당 {avg}~{avg + 1}명 배정")
                        else:
                            st.info(f"👉 조당 {avg}명 배정")

                with c2:
                    responsive_text(f"🎾 경기 규칙", pc_size="20px", mobile_size="16px")
                    v_space(5)
                    default_s_g = saved_cfg.get("s_games", 2)
                    default_d_g = saved_cfg.get("d_games", 1)

                    s_g = st.number_input("단식 게임", min_value=0, max_value=10, value=default_s_g, disabled=not is_admin)
                    d_g = st.number_input("복식 게임", min_value=0, max_value=5, value=default_d_g, disabled=not is_admin)

                    set_options = [2, 3, 4, 5]
                    set_c = st.selectbox("개인전 선승 세트", options=set_options,
                                         index=set_options.index(saved_cfg.get("set_count", 3)) if saved_cfg.get("set_count", 3) in set_options else 1)

                with c3:
                    responsive_text(f"⚙️ 환경 설정", pc_size="20px", mobile_size="16px")
                    v_space(5)
                    default_t = saved_cfg.get("t", 3)
                    t_val = st.number_input("Table_No.", min_value=1, max_value=20, value=default_t, disabled=not is_admin)

                with c4:
                    responsive_text(f"🎲 방식", pc_size="20px", mobile_size="16px")
                    v_space(5)
                    default_draw = saved_cfg.get("draw_method", "AI 선정")
                    draw_options = ["AI 선정", "제비뽑기", "조편성_신청"]
                    draw_index = draw_options.index(default_draw) if default_draw in draw_options else 0

                    draw_method = st.radio("방식", options=draw_options, index=draw_index,
                                           label_visibility="collapsed", disabled=not is_admin,
                                           key="draw_method_radio",
                                           on_change=on_draw_method_change)

                with c5:
                    responsive_text(f"✅ 실행", pc_size="20px", mobile_size="16px")

                        st.session_state.config = {
                            "g": g_val, "t": t_val, "s_games": s_g, "d_games": d_g, "set_count": set_c,
                            "total_g": s_g + d_g, "draw_method": draw_method,

                        st.session_state.config_confirmed = True
                        st.session_state.teams_generated = False
                        save_room_state(st.session_state.room_name)
                        st.toast("✅ 설정이 성공적으로 저장되었습니다!", icon="🎉")
                        time.sleep(0.5)
                        st.rerun()

                    if is_empty_attendees:
                        st.error("참석자가 없어 편성을 시작할 수 없습니다.")

        # ---------------------------------------------------------
        current_radio_val = st.session_state.get("draw_method_radio", "")

        # [안내] 라디오 버튼만 '제비뽑기'로 누르고 아직 확정 버튼을 안 누른 경우
        if not is_config_confirmed and current_radio_val == '제비뽑기':
            v_space(5)
            st.info("👆 설정이 아직 저장되지 않았습니다. 위에서 <strong>'설정 확정 및 편성 시작'</strong> 버튼을 누르면 이곳에 제비뽑기 화면이 나타납니다.")

        # [실행] 설정을 확정했고, 방식이 '제비뽑기'인 경우
        elif is_config_confirmed and "config" in st.session_state and st.session_state.config.get('draw_method') == '제비뽑기':

            # 제비뽑기 진행 중
            if not st.session_state.get('draw_completed', False):
                v_space(5)
                st.markdown("<hr style='margin: 0px; border: 1px solid #ddd;'>", unsafe_allow_html=True)
                v_space(5)

                cfg = st.session_state.config
                attendees = attendees_df.copy()

                if attendees.empty:
                    st.warning("해당 일자에 참석으로 체크된 인원이 없습니다.")
                else:
                    tie_breakers = cfg.get('tie_breakers', {})
                    sorted_members = prepare_draw_data(attendees_df, tie_breakers)

                    group_size = cfg.get('g', 1)
                    group_size = group_size if group_size > 0 else 1
                    total_levels = math.ceil(len(sorted_members) / group_size)

                    if 'draw_level' not in st.session_state:
                        st.session_state.draw_level = 0
                    if 'draw_results' not in st.session_state:
                        st.session_state.draw_results = {}

                    responsive_text(f"📋 제비뽑기 진행 현황", pc_size="20px", mobile_size="16px")
                    v_space(10)

                    current_level = st.session_state.draw_level
                    progress_percent = min(current_level / total_levels, 1.0) if total_levels > 0 else 1.0
                    st.progress(progress_percent, text=f"제비뽑기 진행률: {int(progress_percent * 100)}%")

                    try:
                        with st.spinner("🎲 제비를 섞고 있습니다..."):
                            time.sleep(0.5)

                        render_draw_process(cfg, sorted_members, total_levels, current_level)

                        if current_level >= total_levels:
                            st.session_state.draw_completed = True
                            save_room_state(st.session_state.room_name)

                            if not st.session_state.get('balloons_shown', False):
                                st.balloons()
                                st.session_state.balloons_shown = True
                            st.rerun()

                    except Exception as e:
                        st.error("🚨 제비뽑기 처리 중 예상치 못한 문제가 발생했습니다.")
                        if is_admin:
                            with st.expander("에러 상세 내용 보기"):
                                st.code(str(e))
                        if st.button("🔄 제비뽑기 초기화 및 다시 시도"):
                            reset_draw_state(reset_config=False)
                            st.rerun()

            # 제비뽑기 완료 후
            else:
                v_space(2)
                st.markdown("<hr style='margin: 0px; border: 1px solid #ddd;'>", unsafe_allow_html=True)
                v_space(2)

                st.success("✅ 제비뽑기가 모두 완료되었습니다! 이제 생성된 조 편성을 확인해 주세요.")

                # 🌟 [버그 수정 완료] 관리자라면 언제든 제비뽑기만 다시 할 수 있음
                if is_admin:
                    if st.button("🔄 제비뽑기 다시 하기", type="secondary"):
                        reset_draw_state(reset_config=False)
                        save_room_state(st.session_state.room_name)
                        st.rerun()

>>>>>>>>>>>>>>>>>>>>>>>>
PC에서 파이썬 코드를 직접 구동하는 것과,
Streamlit을 이용해 스마트폰(웹 브라우저)에서 구동할 때의
실행 속도 관점에서의 차이를 이해하려면 가장 먼저 짚고 넘어가야 할 핵심 전제가 있습니다.

바로 "스마트폰에서 파이썬 코드가 직접 연산되는 것이 아니다"라는 점입니다.
Streamlit은 웹 프레임워크이므로,
실제 파이썬 코드(연산, DB/API 통신 등)는 '서버(PC 또는 클라우드)'**에서 실행되고,
스마트폰은 단순히 인터넷을 통해 '결과 화면(UI)만 렌더링'하는 역할을 합니다.

이러한 구조적 차이 때문에 작성해주신 코드 구조를 기준으로
다음과 같은 실행 속도 차이가 발생합니다.

1. Streamlit의 '전체 코드 재실행(Rerun)' 메커니즘 (가장 큰 차이)
PC (일반 Python 스크립트):
코드가 한 번 실행되면 메모리에 상주하며, 특정 이벤트(버튼 클릭 등)가 발생했을 때 해당 함수나 로직만 부분적으로 실행됩니다.

Streamlit (스마트폰):
사용자가 스마트폰 화면에서 버튼을 누르거나, 텍스트를 입력하거나, Tab을 변경할 때마다
파이썬 스크립트가 위에서 아래로(Import부터 끝까지) 매번 다시 실행됩니다.

속도 영향:
작성하신 DB 및 API 연동, 핵심 로직이 사용자가 화면을 조작할 때마다 처음부터 다시 실행되므로
속도가 매우 느려질 수 있습니다.
이를 방지하기 위해 Streamlit에서는
@st.cache_data나 @st.cache_resource 같은 캐싱(Caching) 기술이 필수적입니다.

2. 네트워크 지연 (Network Latency)
PC:
로컬 환경에서 실행되므로 화면에 결과를 띄우는 데 네트워크 지연이 없습니다.
(DB/API 서버와의 통신 시간만 소요됨)

Streamlit (스마트폰):
스마트폰에서 입력 발생 ->
2. 서버로 데이터 전송 ->
3. 서버에서 코드 실행(DB/API 연동 포함) ->
4. 서버에서 스마트폰으로 변경된 화면(HTML/UI) 전송 ->
5. 스마트폰 브라우저가 화면을 그림.

속도 영향:
스마트폰의 모바일 네트워크(LTE/5G/Wi-Fi) 상태와 서버의 위치에 따라
화면이 업데이트되는 반응 속도(UI 렌더링 지연)가 PC에 비해 체감상 훨씬 느립니다.

3. 하드웨어 성능 (컴퓨팅 파워)
PC:
현재 사용 중인 PC의 CPU와 RAM 성능을 100% 활용하여 연산합니다.

Streamlit (스마트폰):
코드가 호스팅되어 있는 '서버의 성능'에 좌우됩니다.
만약 Streamlit Community Cloud 같은 무료 호스팅을 사용한다면,
할당된 CPU/RAM 리소스가 적어 무거운 '핵심 로직'이나 대용량 데이터 처리 시 PC보다 연산 속도 자체가 느려집니다.

4. 작성하신 코드 구조에 따른 속도 최적화 포인트
말씀하신 코드 구조가 Streamlit 환경에서 속도 저하를 겪지 않으려면 다음 부분들이 중요하게 작용합니다.

Import부 & 유틸리티 함수:
매번 재실행되지만 파이썬 자체의 모듈 캐싱 덕분에 큰 속도 저하는 없습니다.

데이터베이스 및 API 연동:
(가장 큰 병목 구간) 스마트폰에서 탭을 바꿀 때마다 DB/API를 다시 호출하면 엄청난 지연이 발생합니다.
반드시 캐싱(@st.cache_data)을 적용해 한 번 불러온 데이터는 서버 메모리에 저장해두고 써야 합니다.

세션 초기화 (Session State):
Streamlit은 매번 코드가 재실행되므로,
이전 상태(입력값, 보관 설정 등)를 기억하려면 st.session_state를 써야 합니다.
세션에 너무 크고 무거운 데이터를 담아두면 서버 메모리를 차지하여 전반적인 속도 저하를 유발할 수 있습니다.

Tab 화면 구성 및 UI 출력:
스마트폰 브라우저는 PC 모니터보다 화면을 그리는(DOM 렌더링) 성능이 낮을 수 있습니다.
한 번에 너무 많은 데이터(수만 줄의 표나 복잡한 차트)를 Tab에 렌더링하려고 하면
스마트폰 브라우저가 버벅거리거나 멈출 수 있습니다.

요약
PC 구동은 "독립 실행형(Local)"이라 즉각적이고 빠르지만,
Streamlit을 통한 스마트폰 구동은 "서버-클라이언트 통신"** 및 "매 상호작용마다 전체 코드 재실행"이라는
특성 때문에 기본적으로 반응 속도가 느립니다.

따라서 스마트폰에서 쾌적하게 구동하려면
1) 무거운 연산과 DB/API 호출의 캐싱(Caching),
2) 세션 스테이트(Session State)를 활용한 불필요한 재연산 방지가 핵심입니다.

Streamlit에서 UI(사용자 인터페이스)를 구성하는 명령어는
매우 직관적이며, 기능별로 다양하게 제공됩니다.
가장 많이 사용되는 대표적인 UI 관련 명령어들을 카테고리별로 정리해 드립니다.

1. 텍스트 및 기본 출력 (Text & Display)
화면에 글자나 데이터를 보여줄 때 사용합니다.

st.title("문자열"): 앱의 메인 제목을 표시합니다.
st.header("문자열"): 큰 제목을 표시합니다.
st.subheader("문자열"): 작은 제목을 표시합니다.
st.text("문자열"): 일반 텍스트를 표시합니다.
st.markdown("문자열"): 마크다운 문법을 적용하여 텍스트를 표시합니다.
st.write(): 텍스트, 데이터프레임, 차트 등 입력된 객체의 타입을 자동으로 인식하여 화면에 출력하는 만능 명령어입니다.

2. 입력 위젯 (Input Widgets)
사용자로부터 데이터를 입력받거나 상호작용할 때 사용합니다.

st.button("버튼 이름"): 클릭할 수 있는 버튼을 생성합니다.
st.text_input("라벨"): 한 줄의 텍스트를 입력받는 창을 만듭니다.
st.text_area("라벨"): 여러 줄의 텍스트를 입력받는 창을 만듭니다.
st.number_input("라벨"): 숫자를 입력받는 창을 만듭니다.
st.slider("라벨", min, max): 범위를 지정할 수 있는 슬라이더 바를 생성합니다.
st.selectbox("라벨", 리스트): 드롭다운 형태의 단일 선택창을 만듭니다.
st.multiselect("라벨", 리스트): 여러 개를 선택할 수 있는 선택창을 만듭니다.
st.checkbox("라벨"): 체크박스를 생성합니다.
st.radio("라벨", 리스트): 여러 옵션 중 하나만 선택하는 라디오 버튼을 만듭니다.
st.file_uploader("라벨"): 파일을 업로드할 수 있는 위젯을 생성합니다.

3. 레이아웃 및 화면 분할 (Layout & Containers)
화면의 배치를 다르게 하거나 요소를 그룹화할 때 사용합니다.

st.sidebar: 좌측 사이드바에 UI 요소를 배치합니다. (예: st.sidebar.selectbox(...))
st.columns(n): 화면을 가로로 n개의 열(단)로 나눕니다.
st.tabs(["탭1", "탭2"]): 여러 개의 탭을 만들어 화면을 전환할 수 있게 합니다.
st.expander("라벨"): 클릭하면 아래로 펼쳐지는 아코디언 형태의 패널을 만듭니다.
st.container(): 여러 UI 요소들을 하나의 블록으로 묶어줍니다.

4. 데이터 표시 (Data Display)
데이터프레임이나 수치를 시각적으로 보여줄 때 사용합니다.

st.dataframe(df): 스크롤 및 정렬이 가능한 동적 테이블을 표시합니다.
st.table(df): 고정된 형태의 정적 테이블을 표시합니다.
st.metric("라벨", value, delta): 주식 가격이나 온도 변화 같은 핵심 지표(KPI)를 표시합니다.

5. 상태 및 알림 메시지 (Status & Feedback)
사용자에게 현재 상태나 경고 등을 알릴 때 사용합니다.
st.success("메시지"): 초록색 성공 알림창
st.error("메시지"): 빨간색 에러 알림창
st.warning("메시지"): 노란색 경고 알림창
st.info("메시지"): 파란색 정보 알림창
st.spinner("메시지"): 작업이 진행되는 동안 로딩 애니메이션을 보여줍니다.
st.progress(숫자): 진행률(0~100)을 나타내는 프로그레스 바를 표시합니다.

Streamlit에서 기본적으로 제공하는 st.success(), st.error(), st.warning(), st.info() 명령어는 앱의 기본 테마를 따르기 때문에 명령어 자체에 색상이나 글자 크기를 직접 지정하는 파라미터는 없습니다.

하지만 **st.markdown**을 활용하면 원하는 색상과 글자 크기를 가진 알림 메시지를 자유롭게 만들 수 있습니다. 크게 두 가지 방법이 있습니다.

방법 1. Streamlit 마크다운 문법 사용 (간단한 텍스트 색상/크기 변경)
Streamlit 1.16.0 버전 이상부터는 마크다운 내에서 간단히 텍스트 색상을 지정할 수 있습니다. 글자 크기는 마크다운의 제목 기호(#)를 사용하여 조절합니다.

Copy
import streamlit as st

# 글자 크기(###)와 텍스트 색상(:green[]) 지정
st.markdown("### :green[✅ 성공적으로 데이터를 저장했습니다!]")
st.markdown("## :red[🚨 치명적인 오류가 발생했습니다!]")
st.markdown("#### :blue[ℹ️ 참고: 업데이트가 예정되어 있습니다.]")
st.markdown("##### :orange[⚠️ 주의: 입력값이 비어있습니다.]")
지원되는 색상: blue, green, orange, red, violet, gray, grey, rainbow
장점: 코드가 매우 간결합니다.
단점: 텍스트 색상만 바뀌며, 기존 st.success처럼 배경색이 있는 박스 형태는 아닙니다.
방법 2. HTML/CSS를 활용한 커스텀 알림창 (배경색, 글자 크기 완벽 제어)
기존의 알림창처럼 배경색이 있는 박스 형태를 유지하면서 글자 크기와 색상을 내 마음대로 바꾸고 싶다면, HTML과 CSS를 작성한 뒤 st.markdown(..., unsafe_allow_html=True)를 사용해야 합니다.

Copy
import streamlit as st

# 1. 커스텀 성공 메시지 (초록색 배경, 큰 글자)
success_html = """
<div style="background-color: #d4edda; color: #155724; padding: 15px; border-radius: 5px; font-size: 24px; font-weight: bold;">
    ✅ 성공: 글자 크기 24px의 커스텀 성공 메시지입니다.
</div>
"""
st.markdown(success_html, unsafe_allow_html=True)

st.write("") # 간격 띄우기

# 2. 커스텀 에러 메시지 (빨간색 배경, 작은 글자)
error_html = """
<div style="background-color: #f8d7da; color: #721c24; padding: 10px; border-radius: 5px; font-size: 14px;">
    🚨 에러: 글자 크기 14px의 커스텀 에러 메시지입니다.
</div>
"""
st.markdown(error_html, unsafe_allow_html=True)

st.write("") # 간격 띄우기

# 3. 커스텀 정보 메시지 (파란색 배경, 중간 글자)
info_html = """
<div style="background-color: #cce5ff; color: #004085; padding: 20px; border-radius: 10px; font-size: 18px;">
    ℹ️ 정보: 패딩(여백)을 넓게 주고 모서리를 더 둥글게 만든 메시지입니다.
</div>
"""
st.markdown(info_html, unsafe_allow_html=True)
CSS 속성 설명:

background-color: 박스의 배경색을 지정합니다. (예: #d4edda)
color: 글자 색상을 지정합니다. (예: #155724)
font-size: 글자 크기를 지정합니다. (예: 24px)
padding: 박스 안쪽의 여백을 지정합니다.
border-radius: 박스 모서리의 둥근 정도를 지정합니다.
이처럼 st.markdown과 unsafe_allow_html=True 옵션을 조합하면 Streamlit의 기본 UI 한계를 넘어 원하는 디자인의 알림창을 완벽하게 구현할 수 있습니다.

로컬 PC에서 streamlit run ... 명령어로 실행하고,
그 PC의 웹 브라우저(예: localhost:8501)를 띄워서 사용하는 경우는
"내 PC가 서버(연산) 역할과 클라이언트(화면 출력) 역할을 동시에 모두 수행하는 상태"가 됩니다.

이전 대화에서 설명해 드린 기준들과 비교했을 때, 다음과 같은 특징을 가집니다.

1. 네트워크 지연 (Network Latency) ➡️ 거의 사라짐 (가장 큰 장점)
스마트폰으로 접속할 때는 인터넷망을 거쳐야 했지만,
로컬 PC에서 실행하면 내부 통신(Localhost)만 일어납니다.

따라서 버튼을 누르거나 탭을 변경할 때,
서버(내 PC)에서 처리된 결과가 브라우저(내 PC)로 전달되는 시간이 거의 0에 가깝습니다.
스마트폰에서 느꼈던 답답한 화면 전환 지연이 사라지고 훨씬 빠르고 즉각적으로 반응합니다.
(단, 코드 내에 있는 외부 DB나 외부 API와 통신하는 시간은 인터넷을 타야 하므로 여전히 소요됩니다.)

2. Streamlit의 '전체 코드 재실행(Rerun)' 메커니즘
➡️ 여전히 똑같이 발생 (주의!)
로컬 PC에서 실행하더라도 Streamlit 프레임워크의 본질적인 작동 방식은 변하지 않습니다.
사용자가 화면에서 무언가 조작(입력, 탭 변경 등)을 할 때마다
Import부부터 끝까지 코드가 매번 다시 실행되는 것은 동일합니다.
따라서 로컬 PC의 성능이 아무리 좋아도,
매번 DB를 다시 읽어오거나 무거운 연산을 처음부터 다시 하게 두면 버벅거릴 수밖에 없습니다.
로컬 환경이라 하더라도 @st.cache_data를 통한 캐싱과 st.session_state 관리는 필수입니다.

3. 하드웨어 성능 (컴퓨팅 파워)
➡️ 내 PC 성능 100% 활용
무료 클라우드 서버 등을 쓸 때 겪는 메모리 부족이나 CPU 성능 제한이 없습니다.
작성하신 '핵심 로직'이나 '데이터 처리' 과정에서
현재 사용 중인 PC의 CPU와 RAM 자원을 온전히 끌어다 쓰기 때문에 연산 속도 자체가 매우 빠릅니다.

4. 화면 렌더링 성능
➡️ PC 브라우저의 우수한 성능 활용
스마트폰의 모바일 브라우저보다 PC의 웹 브라우저(크롬, 엣지 등)가
복잡한 화면(DOM)을 그리는 능력이 훨씬 뛰어납니다.
따라서 수만 건의 데이터가 있는 표(Dataframe)나 복잡한 차트를 화면에 띄울 때,
스마트폰에서는 브라우저가 멈추거나 버벅거리던 현상이 로컬 PC에서는 아주 매끄럽게 처리됩니다.

💡 요약
로컬 PC에서 streamlit run을 실행하는 것은 가장 빠르고 쾌적한 Streamlit 환경입니다.
네트워크 지연이 없고 내 PC의 강력한 성능을 그대로 쓰기 때문입니다.

하지만 일반적인 파이썬 스크립트(한 번 실행 후 메모리 상주)와 달리,
"화면을 조작할 때마다 코드가 처음부터 끝까지 재실행된다"는
Streamlit 고유의 특성은 로컬에서도 그대로 유지됩니다.
따라서 코드 내의 DB/API 연동이나 무거운 로직에 대한 캐싱(Caching) 최적화는
로컬 PC 구동 시에도 반드시 적용되어야 합니다.

Streamlit의 '전체 코드 재실행(Rerun)' 메커니즘에 대해 자세하게 설명해줄래
전체 코드 모두를 재실행하는 건지?
일반적인 생각에 이미 import했거나, 변수 지정 또는 한 번 입력되고, file을 download , upload 완료한 내용
그리고 참석자 확인, 게임구성, 조별 구성은 완료하고, 수정 불가한 조건으로 바꾸고
이후에 일부 화면에서만 data 입력, 연산, 출력하는 과정에 있을 경우
재실행하는 내용에 대해 사용자가 정의하거나 해서
재실행할 떄마다의 속도 저하 문제를 개선할 수 있냐는 질문이야
보다 구체적으로 설명을 해줘

>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>

Streamlit의 '전체 코드 재실행(Rerun)' 메커니즘에 대해
질문하신 상황(참석자 확인, 게임 구성 완료 후 일부 화면만 업데이트 등)에 맞추어
아주 구체적으로 설명해 드리겠습니다.

결론부터 말씀드리면,
"코드는 무조건 위에서 아래로 끝까지 다시 읽지만,
사용자가 정의(캐싱, 세션, 부분 재실행)하여 특정 로직의 '실행(연산)'을
건너뛰거나 특정 구역만 재실행하게 만들 수 있습니다."

질문하신 항목별로 재실행 시 어떤 일이 벌어지는지, 그리고 속도 저하를 어떻게 개선하는지 정리해 드립니다.

1. 질문하신 항목별 재실행(Rerun) 시의 실제 동작

① Import 부
동작: 코드는 다시 읽히지만,
파이썬 자체의 모듈 캐싱 시스템(sys.modules) 덕분에 실제로 라이브러리를 다시 메모리에 적재하지 않습니다.
속도 영향: 없음 (매우 빠름). 신경 쓰지 않으셔도 됩니다.

② 일반 변수 지정
동작:
a = 10이나 data = [] 같은 일반 변수는 재실행될 때마다 초기화(Reset) 됩니다.

속도 영향:
단순 변수 할당은 찰나의 순간이므로 속도 저하는 없지만,
이전 데이터가 날아가는 문제가 발생합니다.

③ File Upload / Download 완료된 내용
동작: st.file_uploader를 통해 업로드된 파일은 Streamlit 서버의 메모리에 유지됩니다.
사용자가 화면을 조작한다고 해서 내 PC/스마트폰에서 파일을 다시 업로드(전송)하지는 않습니다.

속도 영향:
파일 전송 시간은 다시 들지 않지만,
"업로드된 파일을 읽어서 데이터프레임으로 변환하는 코드"**가 있다면
그 코드는 매번 다시 실행되어 속도를 깎아먹습니다. (해결책: 캐싱 필요)

④ 완료 및 수정 불가 상태로 바꾼 설정 (참석자, 게임/조별 구성)
동작:
Streamlit은 "이전에 완료했다"는 사실을 스스로 기억하지 못합니다.
아무런 조치를 취하지 않으면, 탭을 바꾸거나 버튼을 누를 때마다 조별 구성 로직을
처음부터 다시 계산**하려고 합니다.

속도 영향: 매우 큼. 여기서 엄청난 속도 저하와 데이터 꼬임이 발생합니다.

2. 속도 저하 개선 방법 (사용자 정의 최적화)
질문하신 "재실행하는 내용을 사용자가 정의해서 속도 저하를 개선할 수 있느냐?"**에
대한 완벽한 해결책이 Streamlit에 준비되어 있습니다.
다음 3가지 무기를 조합해야 합니다.

💡 해결책 1: st.session_state (상태 기억 및 로직 건너뛰기)
완료된 게임 구성이나 조별 구성을 다시 연산하지 않게 하려면,
**"완료되었다는 상태"를 저장하고 if문으로 로직을 막아버려야 합니다.

# 1. 세션 초기화 (최초 1회만 실행됨)
if 'setup_complete' not in st.session_state:
    st.session_state.setup_complete = False
    st.session_state.team_data = None

# 2. 설정 단계 (완료 전)
if not st.session_state.setup_complete:
    st.subheader("게임 및 조별 구성")
    # ... 데이터 입력 및 조별 구성 로직 ...
    if st.button("구성 완료 및 확정"):
        st.session_state.team_data = "생성된 조별 데이터"
        st.session_state.setup_complete = True
        st.rerun() # 화면 즉시 새로고침

# 3. 설정 완료 후 (수정 불가 상태)
else:
    st.success("조별 구성이 완료되었습니다. (수정 불가)")
    st.write(st.session_state.team_data) # 연산 없이 저장된 결과만 출력
효과:
재실행 시 코드가 위에서 아래로 흐르지만,
if not st.session_state.setup_complete:
조건에 막혀 무거운 조별 구성 로직은 아예 건너뛰게 됩니다.

💡 해결책 2: @st.cache_data (데이터 변환/연산 결과 저장)
업로드한 파일을 읽거나, DB에서 데이터를 가져오는 함수 위에 이 데코레이터를 붙입니다.

@st.cache_data
def load_uploaded_file(file):
    # 엑셀/CSV를 읽는 무거운 작업
    return pd.read_csv(file)

uploaded_file = st.file_uploader("파일 업로드")
if uploaded_file:
    # 재실행되어도, 파일이 바뀌지 않는 한 연산하지 않고 메모리에서 즉시 꺼내옴
    df = load_uploaded_file(uploaded_file)

💡 해결책 3:
@st.fragment (부분 재실행 - ★가장 핵심적인 최신 기능)
질문하신 "이후에 일부 화면에서만
data 입력, 연산, 출력하는 과정"에 완벽하게 부합하는 기능입니다.
(Streamlit 1.37 버전 이상 지원)

특정 함수 위에 @st.fragment를 붙이면,
그 함수 안에서 일어나는 버튼 클릭이나 데이터 입력은 전체 코드를 재실행하지 않고
딱 그 함수 내부만 재실행시킵니다.

st.title("전체 화면 (여기는 다시 실행 안 됨)")
st.write("참석자 및 조별 구성 완료 상태...")

# --- 부분 재실행 구역 정의 ---
@st.fragment
def game_play_section():
    st.subheader("게임 점수 입력 (여기만 재실행됨)")

    # 이 안에서 버튼을 누르거나 값을 입력해도,
    # 파일 업로드나 조별 구성 코드가 있는 바깥쪽은 절대 재실행되지 않음!
    score = st.number_input("1조 점수 입력", value=0)
    if st.button("점수 반영"):
        st.write(f"1조의 현재 점수는 {score}점 입니다.")

# 함수 호출
game_play_section()

🚨 현재 구조의 문제점
API 호출 폭탄 (속도 저하):
만약 load_room_list() 함수에 캐싱(@st.cache_data)이 걸려있지 않다면,
함수가 호출될 때마다 구글 시트(DB)에 접속해서 데이터를 읽어옵니다.
화면을 한 번 클릭할 때마다 DB를 여러 번 조회하게 되어 앱이 심각하게 느려집니다.

불필요한 연산 반복:
캐싱이 걸려있다고 하더라도, 굳이 똑같은 데이터를 여러 함수에서 각자 불러오는 것은 비효율적입니다.

데이터 불일치 위험:
A 함수가 실행될 때와 B 함수가 실행될 때 그 찰나의 순간에 DB가 변경되었다면,
앱 내부에서 데이터가 꼬일 수 있습니다.

💡 개선 방법 (3가지 조합)
가장 이상적인 Streamlit 앱의 구조는 "최상단에서 딱 한 번만 불러오고,
나머지 함수들은 그 데이터를 돌려쓰는 것"입니다.

1단계:
함수 자체에 강력한 캐싱 적용 (필수)

질문자님께서
이미 소프트웨어 공학에서 다루는 핵심적인 평가 기준들을 매우 정확하고 깊이 있게 짚어주셨습니다.
 말씀하신 내용들은 실제 실무에서 S/W 품질을 평가하는 국제 표준(ISO/IEC 25010 등)과
 개발/운영 실무 가이드라인에 모두 포함되는 중요한 항목들입니다.

질문자님의 의견을 바탕으로,
실무에서 사용하는 '소프트웨어 종합 평가 기준'을 7가지 핵심 영역으로 체계화하여 자세히 정리해 드립니다.

1. 사용성 및 UI/UX (Usability)
사용자가 소프트웨어를 얼마나 쉽고 편안하게 사용할 수 있는지를 평가합니다.

직관성 및 학습 용이성:
처음 접하는 사용자도 매뉴얼 없이 쉽게 기능을 파악하고 사용할 수 있는가?

입력 편의성 및 동선 최적화:
PC, 스마트폰 등 디바이스 특성에 맞게 화면이 구성(반응형 등)되었는가?
입력창의 위치, 탭(Tab) 이동 순서, 자동 완성, 드롭다운 등 입력 방식이 사용자의 피로도를 최소화하는가?

접근성:
다양한 환경(브라우저, OS) 및 다양한 사용자(시각/청각적 보조가 필요한 사용자 포함)가 무리 없이 사용할 수 있는가?

2. 기능성 및 성능 (Functionality & Performance)
소프트웨어가 목적에 맞는 기능을 빠르고 정확하게 수행하는지 평가합니다.

정확성 및 적합성:
요구사항에 명시된 기능이 오작동 없이 정확한 결과값을 도출하는가?
응답 속도 (Speed):
사용자의 입력이나 데이터 조회 시 지연 시간(Latency)이 허용 범위 내에 있는가?
자원 효율성:
CPU, 메모리, 네트워크 대역폭 등을 과도하게 점유하지 않고 효율적으로 사용하는가?

3. 신뢰성 및 데이터 무결성 (Reliability & Integrity)
오류 없이 안정적으로 작동하며, 데이터가 훼손되지 않는지를 평가합니다.

데이터 무결성:
데이터의 생성(Create), 보관(Read), 수정(Update), 삭제(Delete) 과정에서
데이터 유실이나 변형이 발생하지 않는가?

결함 허용성 (Fault Tolerance):
특정 기능에 오류가 발생하더라도 시스템 전체가 다운되지 않고 방어할 수 있는가?

예외 처리:
사용자의 잘못된 입력이나 시스템적 오류 발생 시,
적절한 안내 메시지를 띄우고 안전하게 처리하는가?

4. 보안성 (Security)
외부의 위협이나 내부의 실수로부터 시스템과 데이터를 안전하게 보호하는지 평가합니다.

인증 및 권한 관리:
정당한 사용자만 접근할 수 있으며, 직급/역할에 따라 접근할 수 있는
메뉴와 데이터(화면, DB)가 철저히 분리되어 있는가?

데이터 암호화:
비밀번호, 개인정보 등 민감한 데이터가 저장 및 전송될 때 암호화되어 있는가?

취약점 방어:
SQL 인젝션, XSS(크로스 사이트 스크립팅) 등 알려진 웹/앱 보안 취약점에 대한 대비가 되어 있는가?

5. 유지보수성 및 코드 품질 (Maintainability & Code Quality)
개발 이후 코드를 수정하거나 기능을 추가할 때 얼마나 쉽고 안전하게 작업할 수 있는지 평가합니다.

**모듈화 및 재사용성:
반복되는 로직을 함수(Function)나 클래스(Class)로 잘 분리하여 재사용하고 있는가?
(DRY 원칙: Don't Repeat Yourself)

코드 가독성 및 구조화:
코드의 흐름, 배치, 네이밍 규칙(Naming Convention)이 일관되어 다른 개발자가 보아도 이해하기 쉬운가?

결합도와 응집도:
한 부분을 수정했을 때 다른 부분에 미치는 영향(Side-effect)이 최소화되도록 설계되었는가?

이력 관리 (Version Control):
Git, SVN 등을 활용하여 소스 코드의 변경 이력이 체계적으로 관리되고,
문제 발생 시 롤백(Rollback)이 용이한가?

6. 운영성 및 데이터 관리 (Operability & Data Management)
시스템을 지속적으로 운영하고 데이터를 축적/관리하는 능력을 평가합니다.

백업 및 복구 체계 (Backup & Recovery):
DB 삭제, 하드웨어 장애, 랜섬웨어 등에 대비해 주기적인 자동 백업이 이루어지며,
유사시 목표 시간 내에 복구할 수 있는가?

모니터링 및 로깅:
시스템의 상태, 에러 발생 내역, 사용자의 주요 작업 이력(Audit Log)이 기록되어 문제 추적이 가능한가?

확장성 (Scalability):
향후 데이터가 대량으로 축적되거나 사용자가 급증할 때, 서버나 DB를 쉽게 확장할 수 있는 구조인가?

7. 기술 적합성 및 리스크 관리 (Technical Appropriateness)
도입한 기술과 툴을 얼마나 잘 활용하며, 그에 따른 부작용을 어떻게 통제하는지 평가합니다.

프레임워크/툴 활용도:
적용된 언어나 프레임워크(Tool)가 제공하는 내장 기능과 라이브러리를 억지스럽지 않게,
기술적 특성에 맞게 100% 활용하고 있는가?

사이드 이펙트(Side-effect) 제어:
새로운 기술이나 패치를 적용할 때 기존 시스템에 미치는 영향을
사전에 테스트(단위 테스트, 통합 테스트)하고 보완할 프로세스가 있는가?

💡 요약하자면:
실제 기업이나 공공기관에서는 위 항목들을 바탕으로 **'소프트웨어 품질 평가 체크리스트'를 만듭니다.

이 틀을 바탕으로 평가하고자 하는 S/W의 특성에 맞춰 항목별 가중치(점수)를 부여하여
평가표를 구성하시면 완벽한 종합 평가 기준이 될 것입니다.

📊 소프트웨어 종합 평가 및 개선 방안

1. 사용성 및 UI/UX (Usability) - [13점 / 15점]
평가:
탭(Tabs)을 활용한 논리적인 화면 분할, 반응형 텍스트(responsive_text),
Toast 및 Balloon 애니메이션을 통한 시각적 피드백 등 사용자 편의성을 깊이 고민한 흔적이 보입니다.

개선 필요 사항:
모바일 환경 최적화:
st.data_editor나 넓은 DataFrame(매트릭스 스코어보드 등)은
모바일 화면에서 가로 스크롤이 발생하여 조작이 불편할 수 있습니다.
모바일 접속 시에는 요약된 카드 형태로 보여주거나,
컬럼 수를 줄이는 반응형 로직이 추가되면 좋습니다.

2. 기능성 및 성능 (Functionality & Performance) - [13점 / 15점]
평가:
조 편성(AI, 제비뽑기), 대진표 자동 생성, 스코어 계산 등
복잡한 비즈니스 로직이 잘 구현되어 있습니다.
캐싱과 Fragment를 통한 성능 최적화도 훌륭합니다.

개선 필요 사항:
동기적 API 호출 지연:
구글 시트(gspread, st.connection)에 데이터를 읽고 쓰는 작업이 동기적(Synchronous)으로 처리되고 있습니다.
클라우드 저장 버튼을 누르면 완료될 때까지 UI가 멈춥니다(Spinner가 돌지만).
비동기 처리(asyncio)를 도입하거나, 백그라운드 스레드에서 저장되도록 개선하면 체감 속도가 훨씬 빨라집니다.

3. 신뢰성 및 데이터 무결성 (Reliability & Integrity) - [11점 / 15점]
평가:
로컬 JSON 파일(room_states.json, league_history.json)을 활용해
구글 시트 오류 시에도 데이터를 복구할 수 있는 백업 체계를 갖춘 점은 매우 훌륭합니다.

개선 필요 사항:
포괄적인 예외 처리 (Broad Exception):
코드 곳곳에 except: 구문이 사용되었습니다.
(예: extract_busu 함수, 파일 로드 부분 등). 이는 예상치 못한 치명적 오류까지 숨겨버려 디버깅을 어렵게 만듭니다.
except ValueError:, except FileNotFoundError: 등 구체적인 에러를 명시해야 합니다.

동시성 문제 (Race Condition):
여러 사용자가 동시에 접속하여 JSON 파일에 쓰기(json.dump)를 시도할 경우 파일이 깨질(Corrupted) 위험이 있습니다.
filelock 라이브러리 등을 사용하여 파일 쓰기 시 Lock을 걸어주어야 합니다.

4. 보안성 (Security) - [11점 / 15점]
평가: st.secrets를 활용해 민감 정보를 분리하고,
쿠키를 암호화(EncryptedCookieManager)한 점은 보안 기본기를 잘 지킨 것입니다.

개선 필요 사항:
**단순 해시 사용: hashlib.sha256(password.encode()).hexdigest() 방식으로 비밀번호를 암호화하고 있습니다.
이는 레인보우 테이블(Rainbow Table) 공격에 취약합니다.
보안성을 높이려면 솔트(Salt)**를 추가하거나, bcrypt 라이브러리를 사용하는 것이 표준입니다.

개선 코드 예시: hashlib.pbkdf2_hmac('sha256', password.encode(), salt, 100000)

5. 유지보수성 및 코드 품질 (Maintainability & Code Quality) - [10점 / 15점]
평가: 주석이 매우 상세하고 변수명이 직관적입니다.

개선 필요 사항:
모놀리식(Monolithic) 구조:
하나의 파일에 UI 렌더링, 데이터베이스 통신, 비즈니스 로직(대진표 짜기 등)이 모두 섞여 있어 코드가 매우 깁니다.

관심사 분리 (Separation of Concerns):
코드를 여러 모듈로 분리해야 합니다.

database.py (구글 시트 및 JSON 저장 로직)
logic.py (조 편성, 대진표 생성, 점수 계산 로직)
ui_components.py (Fragment 및 화면 그리기 로직)
main.py (Streamlit 탭 구성 및 메인 실행)

6. 운영성 및 데이터 관리 (Operability & Data Management) - [12점 / 15점]
평가:
구글 시트를 DB로 활용하여 관리자가 직접 데이터를 수정하기 쉽도록 한 점은 실제 운영 환경을 잘 고려한 아키텍처입니다.

개선 필요 사항:
로깅(Logging) 부재:
에러가 발생했을 때 st.error나 print로만 출력하고 있습니다.
시스템이 지속적으로 운영되려면 Python의 logging 모듈을 사용하여
에러 발생 시간, 발생 위치, 사용자 액션 등을 파일(app.log)로 남겨야 문제 추적(Audit)이 가능합니다.

7. 기술 적합성 및 리스크 관리 (Technical Appropriateness) - [8점 / 10점]
평가:
빠른 사내/동호회용 웹앱 개발을 위해 Streamlit을 선택한 것은 매우 적절합니다.
Pandas를 활용한 데이터 매트릭스 연산도 훌륭합니다.

개선 필요 사항:
사이드 이펙트 제어 (테스트 코드):
get_matches (대진표 생성)나 update_cumulative_record (전적 업데이트) 같은 핵심 로직은
버그가 발생하면 대회 운영에 치명적입니다.
pytest 등을 활용하여 해당 함수들에 대한 단위 테스트(Unit Test) 코드를 별도로 작성하여 검증하는 과정이 필요합니다.

💡 주요 개선 내용 요약
1. 보안성 (Security) [최우선]

문제점:
hashlib.sha256()을 단순 사용하면 '레인보우 테이블(Rainbow Table)' 해킹 공격에 매우 취약합니다.
또한 쿠키 비밀번호에 하드코딩된 기본값(default_fallback_password)이 노출되어 있습니다.

개선:
hmac 라이브러리와 st.secrets의 Salt(소금치기) 값을 결합하여 암호화를 강화했습니다.
하드코딩된 비밀번호 폴백을 제거하여 보안을 강제했습니다.

2. 신뢰성 및 데이터 무결성 (Reliability & Integrity) [필수]

문제점:
save_room_state 등에서 JSON 파일에 open(..., "w")로 직접 덮어쓰기를 하고 있습니다.
 만약 저장하는 도중 서버가 다운되거나 에러가 나면 기존 파일 내용이 전부 날아가고 빈 파일이 되어 데이터가 영구 유실됩니다.

개선:
원자적 쓰기(Atomic Write) 방식을 도입했습니다.
임시 파일(.tmp)에 먼저 데이터를 안전하게 쓴 뒤,
성공했을 때만 원본 파일을 바꿔치기(os.replace)하여 데이터 증발을 원천 차단했습니다.

3. 운영성 및 데이터 관리 (Operability & Data Management) [유지]

문제점:
에러가 발생하거나 예외 상황일 때 print()를 사용하고 있습니다.
서버(클라우드) 환경에서는 print() 로그가 휘발되어 관리자가 문제 원인을 추적할 수 없습니다.

개선:
파이썬 표준 logging 모듈을 도입하여,
시스템의 주요 동작과 에러 발생 내역을 시간과 함께 체계적으로 기록하도록 변경했습니다.

4. 기능성 및 성능 (Functionality & Performance) [핵심]

문제점:
구글 시트 API 호출 시 일시적인 네트워크 오류로 앱이 멈출 수 있습니다.

개선:
데이터베이스 업데이트(conn.update) 시 예외 처리(try-except)와 로깅을 강화하여,
네트워크 지연 시에도 앱이 튕기지 않도록 방어 로직을 추가했습니다.

5. 사용성 및 UI/UX (Usability) [가치 창출]

개선:
사용자가 알 필요 없는 내부 시스템 에러(예: 매트릭스 인덱스 오류)는
화면에 경고창(st.warning)으로 부드럽게 띄우고, 상세 에러는 백그라운드 로그로 남기도록 분리했습니다.

6. 기술 적합성 및 리스크 관리 (Technical Appropriateness) [안정]

개선:
Streamlit Cloud 등에서는 로컬 파일(.json)이 재부팅 시 초기화되는 부작용(Side-effect)이 있습니다.
이를 대비해 파일 입출력 시 FileNotFoundError 등을 완벽히 방어하도록 예외 처리를 보강했습니다.

💻 수정된 전체 코드
(※ 요청하신 대로 '유지보수성(함수/클래스 분리 등)'은 보류하고,

올려주신 코드는 Streamlit을 기반으로 한 '리그 스코어보드 및 결과 관리 시스템'입니다.

💡 주요 변경 내용 핵심 설명
1. 중복 저장 방지 로직 (데이터 무결성 및 용량 최적화)
**이전 문제:
사용자가 '저장' 버튼을 여러 번 누르면,
변경된 점수가 없는데도 똑같은 데이터가 시간만 다르게 계속 저장되어 용량을 차지하고 목록이 지저분해졌습니다.

변경 내용:
st.session_state.last_saved_matrix라는 변수를 만들어
**'가장 마지막에 저장한 스코어보드 상태'를 기억하게 했습니다.
현재 화면의 스코어와 마지막 저장본을 비교(equals)하여,
완전히 똑같으면 저장을 막고 경고 메시지를 띄웁니다.

2. @st.fragment 도입 (속도 개선 및 깜빡임 방지)
이전 문제:
점수를 하나 입력하거나 글자 크기를 바꿀 때마다 웹페이지 전체가 새로고침(Rerun)되어
화면이 깜빡이고 속도가 느려졌습니다.

변경 내용:
스코어보드를 그리는 함수(draw_summary_table)와
점수를 입력하는 함수(render_manual_score_input) 위에 @st.fragment라는 장식을 붙였습니다.
이를 통해 전체 화면은 가만히 두고,
점수가 바뀌는 해당 표 부분만 빠르게 부분 새로고침되도록 개선했습니다.

3. st.data_editor 캐시 강제 초기화 (버그 해결)
이전 문제:
표(Data Editor)에서 점수를 직접 더블클릭해서 수정하고 '적용'을 눌렀을 때,
화면에 이전 점수가 그대로 남아있는 버그(캐시 문제)가 발생할 수 있습니다.

변경 내용:
표에 key="main_matrix_editor"라는 이름표를 붙여주고,
변경사항을 DB에 저장한 직후 del st.session_state["main_matrix_editor"] 코드를 실행하여
에디터가 가지고 있던 임시 기억(캐시)을 강제로 지워버립니다.
이렇게 하면 항상 최신 점수만 깔끔하게 표시됩니다.

4. 이미 끝난 경기 '입력창 잠금' 기준 변경
이전 문제:
A선수와 B선수의 점수 칸이 단순히 '비어있지 않다'는 이유만으로
입력창이 잠겨버리는 오류가 있었습니다. (예: 0점 처리된 경우 등)

변경 내용:
float(val_a) + float(val_b) > 0 조건을 추가했습니다.
즉, 두 선수의 점수 합이 0보다 클 때(실제로 경기를 치러서 점수가 났을 때)만 경기가 끝난 것으로 간주하고
입력창을 비활성화(잠금)하도록 정확도를 높였습니다.

💻 수정 및 정리된 핵심 코드
위의 설명이 적용된 전체 흐름을 이해하기 쉽도록, 불필요한 부분은 줄이고 핵심 로직 위주로 정리한 코드입니다.

Streamlit의 표 편집기(st.data_editor)는 엑셀처럼
'특정 셀(Cell)만 클릭을 막는 기능'을 기본적으로 제공하지 않습니다.
따라서 화면에 보여주는 데이터와 실제 저장되는 데이터를 분리하는 방식으로 이 문제를 깔끔하게 해결했습니다.

💡 주요 변경 포인트
화면 표시 전처리 (view_df):
표를 화면에 그리기 전에, 자기 자신과의 매칭(대각선)은 "None"으로,
아직 입력되지 않은 빈칸은 "-"로 변환하여 보여줍니다.

입력 원천 차단 (비활성화 효과):
사용자가 실수로 "None" 칸을 다른 숫자로 수정하고 [적용]을 누르더라도,
저장 로직에서 if r == c: continue를 통해 자기 자신과의 매칭 점수는
아예 무시하도록 처리하여 비활성화와 동일한 효과를 냈습니다.

데이터 복원:
"-"나 "None" 같은 문자열은 저장될 때 다시 빈칸(np.nan)으로 안전하게 변환되어
통계 계산에 오류가 없도록 했습니다.

금방 다시 돌아온 경우 (스마트폰 메모리에 여유가 있을 때)

상태 유지:
화면이 새로고침되지 않고, 나가기 직전의 화면이 그대로 유지됩니다.
데이터 보존:
입력하던 텍스트나 Streamlit의 세션 상태(st.session_state) 데이터도 그대로 남아있어 이어서 작업할 수 있습니다.

2. 시간이 꽤 지나거나 다른 무거운 앱을 실행한 후 돌아온 경우 (메모리 부족 시)
새로고침 발생:
스마트폰 OS(iOS, Android)는 배터리와 메모리 절약을 위해
백그라운드에 오래 있거나 메모리를 차지하는 브라우저 탭의 프로세스를 강제로 종료합니다.
다시 돌아오면 웹페이지가 자동으로 새로고침됩니다.

데이터 초기화 (주의 ⚠️):
Streamlit은 브라우저와 서버 간의 실시간 연결(WebSocket)을 기반으로 작동합니다.
페이지가 새로고침되어 연결이 끊겼다가 다시 맺어지면 새로운 사용자가 접속한 것으로 인식합니다.
따라서 st.session_state에 임시로 저장해둔 데이터나 입력 중이던 내용이
모두 초기화(증발)되고 처음 화면으로 돌아가게 됩니다.

(데이터 증발 방지 팁) 만약 모바일 환경에서 사용자가 나갔다 와도(새로고침 되어도)

새로고침 되어도 URL은 유지되므로 이전 상태로 복구할 수 있습니다.

브라우저 쿠키/로컬 스토리지 사용:
streamlit-cookies-manager나 streamlit-local-storage 같은
외부 라이브러리를 사용해 스마트폰 브라우저 자체에 데이터를 저장합니다.

데이터베이스(DB) 연동:
중요한 데이터는 입력 즉시 외부 DB(Firebase, Supabase, MySQL 등)에 저장하여,
새로고침 시 DB에서 다시 불러오도록 설계합니다.

적합한 데이터:

별도의 서버(DB)를 구축할 필요 없이 사용자의 스마트폰 브라우저 자체에 데이터를 저장하므로 서버 부담이 없습니다.

적합한 데이터:
다크모드/라이트모드 설정, 작성 중이던 긴 텍스트(임시저장 기능), 장바구니 내역 등

3. 절대 날아가면 안 되는 중요한 데이터라면?

장점:
가장 안전하고 확실한 방법입니다. 앱이 강제 종료되거나, 폰을 바꾸거나, PC로 접속해도 데이터가 그대로 유지됩니다.

적합한 데이터:
회원 정보, 결제 내역, 최종 제출해야 하는 설문조사나 폼 데이터 등

단점: DB(Firebase, Supabase 등)를 세팅해야 하므로 개발 난이도가 가장 높습니다.

만약 완성도 높은 모바일용 Streamlit 앱을 만들고 싶으시다면, 1번과 3번을 조합하는 것이 가장 좋습니다.

UI 상태 (현재 어떤 탭을 보고 있는지 등):
URL 파라미터로 관리하여 새로고침 되어도 원래 보던 화면이 뜨게 만듭니다.

사용자 입력 데이터:
사용자가 값을 입력할 때마다 DB에 수시로 자동 저장(Auto-save)되도록 코드를 작성하여,
앱이 튕기더라도 다시 들어왔을 때 DB에서 마지막 상태를 불러오게 하는 것이 가장 완벽한 해결책입니다.

네, 질문하신 두 가지 모두 충분히 가능하며, 모바일 환경을 고려한다면 반드시 적용해야 하는 필수적인 조치입니다.

질문하신 내용을 바탕으로 어떻게 해결할 수 있는지, 그리고 이전 대화에서 언급한 '데이터 증발 방지 대책'을 어떻게 코드에 녹여낼 수 있는지 구체적으로 설명해 드리겠습니다.

1. 비활성화 풀었을 때 이전 데이터(선수, 결과) 다시 보여주기
현재 다시 선택했을 때 데이터가 초기화되는 이유는, Streamlit이 화면을 다시 그릴 때 위젯(Selectbox, Radio 등)의 상태를 기억하지 못하고 기본값으로 되돌리기 때문입니다.

해결책: st.session_state를 위젯의 기본값(value 또는 index)으로 설정 각 경기의 입력 결과를 st.session_state에 딕셔너리 형태로 저장해두고, 화면을 그릴 때 이 저장소에 값이 있으면 그 값을 위젯의 기본값으로 띄워주도록 코드를 수정해야 합니다.

2. 모바일 '새로고침(튕김)' 상황에 대비한 대책 적용
현재 "전체 경기가 완료된 다음에 저장"하는 방식은 모바일 환경에서 가장 위험한 구조입니다. 3경기까지 입력했는데 폰에서 잠시 카카오톡을 보고 왔더니 새로고침되어 1~3경기 데이터가 모두 날아갈 수 있습니다.

해결책: 개별 경기 즉시 저장(DB 연동) + 초기 로드 시 복구 이전 대화에서 추천해 드린 '3번(DB 연동) + 1번(상태 유지)' 조합을 적용해야 합니다. 전체 완료 후 저장이 아니라, 1경기가 끝날 때마다 즉시 DB(또는 파일/로컬스토리지)에 저장하도록 구조를 바꿔야 합니다.

제시해주신 코드를 바탕으로, 이전 대화에서 강조했던 "비활성화(또는 다른 행 클릭) 후 다시 돌아왔을 때 이전 입력 데이터 유지" 및 "모바일 새로고침 대비 즉시 저장" 로직을 적용하여 코드를 수정했습니다.

가장 핵심적인 변경 사항은 상세 결과를 입력하는 @st.fragment def render_match_details 내부입니다.

💡 주요 수정 포인트
st.session_state.match_details 도입: 각 경기(m_idx)별로 선택했던 선수 이름, 승/패 결과, 스코어를 딕셔너리 형태로 저장하는 공간을 만들었습니다.
위젯 index 바인딩: selectbox와 radio를 그릴 때, 저장된 데이터가 있다면 해당 값의 인덱스를 찾아 기본값(index)으로 세팅하도록 수정했습니다.
즉시 저장 로직 강화: [저장] 버튼을 누를 때 점수판(matrix)뿐만 아니라, 화면 UI 상태(match_details)도 세션에 즉시 저장되도록 구성했습니다.
아래 수정된 코드를 확인해 주세요. (기존 코드의 @st.fragment def render_match_details 부분을 아래 코드로 교체하시면 됩니다.)

이 앱의 제비뽑기는 '완전 무작위(랜덤)'가 아니라 '실력(부수) 밸런스를 맞추기 위한 레벨별 추첨 방식'을 기준으로 진행됩니다.

코드에 나타난 구체적인 제비뽑기 기준과 순서는 다음과 같습니다.

1. 참석자 필터링

df[df[col_date] == 'Y']
해당 경기 일자에 참석('Y')한다고 체크된 인원만 제비뽑기 대상이 됩니다.
2. 1차 정렬 기준: 실력 (부수)

attendees['부수_숫자'] = attendees['부수'].apply(extract_busu)
참가자들의 '부수(실력 등급)'에서 숫자를 추출하여 오름차순으로 정렬합니다.
즉, 실력이 가장 높은 사람부터 낮은 사람 순(또는 그 반대)으로 한 줄로 세우는 과정입니다.
3. 2차 정렬 기준: 동점자 랜덤 섞기 (Tie-breaker)

attendees['Random'] = attendees['이름'].map(cfg['tie_breakers'])
부수가 같은 사람들(예: 5부 3명)이 있을 경우, 미리 부여된 랜덤 값(tie_breakers)을 기준으로 순서를 무작위로 섞습니다.
4. 레벨(그룹) 분할

total_levels = math.ceil(len(sorted_members) / cfg['g'])
정렬된 전체 명단을 설정된 '조당 인원수(g)'로 나누어 여러 개의 레벨(그룹)로 쪼갭니다.
(예: 총 20명이고 조당 4명씩 배정한다면, 1~4등은 1레벨, 5~8등은 2레벨... 식으로 묶임)
💡 결론적으로 어떤 방식인가요?

이 방식은 탁구나 배드민턴 같은 동호회 대회에서
가장 많이 쓰는 "각 조의 실력 밸런스를 맞추기 위한 제비뽑기"입니다.

전체 인원을 한 통에 넣고 마구잡이로 뽑는 것이 아니라,
비슷한 실력(부수)을 가진 사람들끼리 묶어놓고 그 안에서 제비를 뽑아 각 조로 흩어지게 만드는 기준을 사용하고 있습니다.
이렇게 하면 특정 조에 고수들만 몰리거나 하수들만 몰리는 현상을 방지할 수 있습니다.

제시해주신 아이디어는 각 조의 실력 밸런스를 훨씬 더 정교하게 맞출 수 있는 매우 합리적인 방법입니다!

말씀하신 대로 같은 부수 내에서 부수_조정 숫자가 낮은 사람을 상위 랭커로 취급하도록 정렬 기준을 추가했습니다.

이제 정렬(순위) 기준은 다음과 같이 3단계로 작동합니다.

1차 기준: 부수_숫자 (낮을수록 고수)
2차 기준 (추가): 부수_조정 (같은 부수일 때, 낮을수록 고수)
3차 기준: Random (부수와 부수_조정까지 모두 같을 때 무작위 섞기)
아래는 해당 조건이 추가된 수정 코드입니다. 데이터에 빈칸이나 문자가 섞여 있을 경우를 대비해 안전하게 숫자로 변환하는 방어 로직도 함께 추가했습니다.
